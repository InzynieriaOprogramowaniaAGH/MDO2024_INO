pipeline {
  agent any
  stages {
    stage('Build Image') {
      steps {
        script {
          sh 'docker build -t bartekkep/project:${DOCKER_TAG} -f Dockerfile.build .'
        }

      }
    }

    stage('Verify Image') {
      steps {
        script {
          sh 'docker images'
        }

      }
    }

    stage('Build Project') {
      steps {
        script {
          sh '''
docker run --rm -v ${WORKSPACE}:/app -w /app bartekkep/project:${DOCKER_TAG} \
sh -c "mkdir -p build && cd build && cmake .. && make"
'''
        }

        archiveArtifacts(artifacts: 'build/**', allowEmptyArchive: true)
      }
    }

    stage('Test') {
      steps {
        script {
          sh 'docker build -t bartekkep/test_stage:${DOCKER_TAG} -f Dockerfile.test .'

          sh '''
docker run --rm -v ${WORKSPACE}:/app -w /app/build bartekkep/test_stage:${DOCKER_TAG} \
sh -c "ctest --output-on-failure"
'''
        }

      }
    }

    stage('Deploy') {
      steps {
        script {
          sh '''
docker run -d --name app_container -p ${DEPLOY_PORT}:8081 bartekkep/project:${DOCKER_TAG}
'''
        }

      }
    }

    stage('Publish') {
      steps {
        script {
          withCredentials([usernamePassword(credentialsId: 'dockerhub', usernameVariable: 'DOCKERHUB_USERNAME', passwordVariable: 'DOCKERHUB_PASSWORD')]) {
            sh 'docker login -u $DOCKERHUB_USERNAME -p $DOCKERHUB_PASSWORD'
            // Wypychanie obrazu z poprawnym tagiem
            sh 'docker push bartekkep/project:${DOCKER_TAG}'
          }
        }

      }
    }

  }
  environment {
    DOCKER_TAG = "${env.BUILD_NUMBER}"  // Automatyczne tagowanie obrazów według numeru builda
    DEPLOY_PORT = '8081'
  }
  post {
    always {
      sh 'docker container prune -f'
      sh 'docker image prune -f'
    }

  }
}
